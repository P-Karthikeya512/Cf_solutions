import sys
input = sys.stdin.readline
 
def binpow(base, exp):
    result = 1
    while exp > 0:
        if exp % 2 == 1:
            result *= base
        base *= base
        exp //= 2
    return result
 
def binpowmo(base, exp, mod):
    result = 1
    base %= mod
    while exp > 0:
        if exp & 1:
            result = result * base % mod
        base = base * base % mod
        exp >>= 1
    return result
 
def modinv(a, mod):
    return binpowmo(a, mod - 2, mod)  # mod must be prime
 
def sieve(N):
    is_prime = [True] * (N + 1)
    is_prime[0] = is_prime[1] = False
    for i in range(2, int(N**0.5) + 1):
        if is_prime[i]:
            for j in range(i * i, N + 1, i):
                is_prime[j] = False
    return is_prime
 
def solve():
    n, k = map(int, input().split())
    a = list(map(int, input().split()))
    b = list(map(int, input().split()))
    
    sum_val = -1
    for i in range(n):
        if b[i] != -1:
            if sum_val == -1:
                sum_val = a[i] + b[i]
            elif sum_val != a[i] + b[i]:
                print(0)
                return
 
    if sum_val == -1:
        mx = max(a)
        mn = min(a)
        print(k - (mx - mn) + 1)
        return
 
    for i in range(n):
        if a[i] > sum_val or (sum_val - a[i]) > k:
            print(0)
            return
 
    print(1)
 
def main():
    t = int(input())
    for _ in range(t):
        solve()
 
if __name__ == "__main__":
    main()
